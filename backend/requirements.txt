# Every dependency here is justified. Anything not listed is not needed.

# Postgres driver. v3 specifically: server-side parameter binding, the
# ClientCursor used to render auditable SQL for the inspector, and typed
# exceptions (QueryCanceled) that let a timeout be reported honestly rather
# than mistaken for an empty result.
psycopg[binary,pool]>=3.2

# Tool input models. Validation happens BEFORE any SQL is built, so an
# unknown dimension or an injected identifier is rejected at the boundary.
pydantic>=2.7

# Tests. They run against the live database -- no fabricated fixtures.
pytest>=8.0
