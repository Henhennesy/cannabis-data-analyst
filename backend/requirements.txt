# The full local development set: everything the app needs at runtime, plus
# the test runner and the data pipeline.
#
#   pip install -r backend/requirements.txt      <- local venv, unchanged
#
# The container installs requirements-app.txt instead, which is the runtime
# subset. Keeping the runtime list in one file and including it here means the
# two cannot drift: there is only one place a serving dependency is declared.

-r requirements-app.txt

# --- Tests -----------------------------------------------------------------
# They run against the live database -- no fabricated fixtures.
pytest>=9,<10

# fastapi.testclient requires httpx. It also arrives transitively via anthropic,
# but the tests depend on it directly, so it is declared directly. Also used by
# the probe scripts in scripts/.
httpx>=0.28,<1

# --- Data pipeline only (data_cleaning_tool/), not the running app ---------
# DuckDB does the cleaning and the annual aggregation; openpyxl reads the
# .xlsx exports BDSA sometimes sends instead of CSV.
duckdb>=1.5,<2
openpyxl>=3.1,<4

# Deliberately NOT used:
#   redis          -- SQLite on disk covers a single-user cache
#   sqlalchemy     -- every query is hand-written SQL that must be auditable
#   pandas         -- rows go straight to JSON; no dataframe layer needed
#   python-dotenv  -- a handful of variables, read by a dozen lines in config.py
